  /** Total labor slots occupied by all placed buildings (constructing or working). */
  getEmployedLabor(): number {
    let total = 0;
    for (const b of this.state.buildings) {
      const def = getBuildingDef(b.defId);
      total += def?.cost.people ?? 0;
    }
    return total;
  }

  /** Idle labor = total population minus employed, clamped to zero. */
  getIdleLabor(): number {
    return Math.max(0, (this.state.resources['people'] ?? 0) - this.getEmployedLabor());
  }
